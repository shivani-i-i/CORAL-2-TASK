%% 1. Introduction
\section{Introduction}

This report documents the execution and results of ten benchmarks from the CORAL-2 (Collaboration Oak Ridge Argonne Livermore 2) suite, originally developed by Lawrence Livermore National Laboratory (LLNL) for large-scale HPC system procurement.

All benchmarks were compiled from source and executed on Google Colab (single-node, single-CPU environment) using OpenMPI 4.1.2. Each benchmark was run a minimum of 10 times to capture variance.

%% 3. Application Configurations
\section{Application Configurations}

\subsection{1. STREAM}
Memory bandwidth benchmark. Measures sustainable memory bandwidth (MB/s) for four kernels: Copy, Scale, Add, and Triad. Run with default array sizes on the system's available RAM.

\subsection{2. AMG}
Algebraic MultiGrid solver. Figure of Merit (FOM) is nnz$\cdot$iter/sec. Run with default problem size on 1 MPI rank.

\subsection{3. Kripke}
Deterministic neutron transport (Sn sweep) mini-app. FOM is unknowns/sec/iter. Run with default spatial and angular decomposition on 1 MPI rank.

\subsection{4. PENNANT}
Unstructured mesh hydrodynamics mini-app. Measures hydro cycle wall time in seconds. Run with the \texttt{sedov} test problem (small mesh).

\subsection{5. Nekbone}
Spectral element CFD proxy app. Measures execution time (seconds) and floating-point throughput (MFLOPS). Run with a small 3D mesh on 1 MPI rank.

\subsection{6. HACC (Hardware Accelerated Cosmology Code)}

\textbf{Version:} HACC\_1\_7\\
\textbf{Build flags:}
\begin{verbatim}
HACC_MPI_CXX=mpicxx, HACC_MPI_CC=mpicc
HACC_MPI_CFLAGS=-O3, HACC_MPI_CXXFLAGS=-O3
FFTW_HOME=/usr, DFFTW3 enabled
\end{verbatim}

\textbf{Key input parameters (indat\_clean):}
\begin{verbatim}
ng  = 64         (grid points per dimension, reduced from 512)
np  = 64         (particles per dimension, reduced from 1024)
nsteps = 20
zin = 300.0, zfin = 0.0
sigma8 = 0.80
box size = 64.0 h^{-1}Mpc
MPI ranks = 1
\end{verbatim}

\subsection{7. LAMMPS (stable\_29Aug2024)}

\textbf{Build:}
\begin{verbatim}
cmake ../cmake -DCMAKE_BUILD_TYPE=Release
  -DBUILD_MPI=yes -DPKG_MANYBODY=yes -DPKG_MOLECULE=yes
\end{verbatim}

\textbf{Simulation input (Lennard-Jones liquid):}
\begin{verbatim}
units       lj
atom_style  atomic
lattice     fcc 0.8442
region      box block 0 10 0 10 0 10
pair_style  lj/cut 2.5
pair_coeff  1 1 1.0 1.0 2.5
fix         1 all nve
run         1000
\end{verbatim}

\subsection{8. Quicksilver}

\textbf{Version:} commit eb68bb8 (Aug 2023)\\
\textbf{Build:} HIP/GPU flags removed; compiled with \texttt{mpicxx -DHAVE\_MPI -O2}.

\textbf{Run parameters:}
\begin{verbatim}
-X 10 -Y 10 -Z 10   (mesh dimensions)
-x 10 -y 10 -z 10   (spatial decomposition)
-n 10000             (particles)
-N 10                (cycles)
MPI ranks = 1
\end{verbatim}

\subsection{9. QMCPACK}

\textbf{Version:} latest main (cloned May 21, 2026)\\
\textbf{Build:}
\begin{verbatim}
cmake .. -DCMAKE_C_COMPILER=mpicc
         -DCMAKE_CXX_COMPILER=mpicxx
         -DQMC_MPI=1 -DQMC_OMP=0
\end{verbatim}

\subsection{10. IOR}

\textbf{Build:}
\begin{verbatim}
./bootstrap && ./configure CC=mpicc && make -j$(nproc)
\end{verbatim}

\textbf{Run parameters:}
\begin{verbatim}
-t 1m   (transfer size: 1 MB)
-b 16m  (block size: 16 MB)
-s 16   (segments)
MPI ranks = 1
\end{verbatim}
